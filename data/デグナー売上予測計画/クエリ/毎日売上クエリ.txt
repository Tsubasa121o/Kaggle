/*
Purpose:
- Target JAN codes that start with 452
- Aggregate daily sales by customer
Granularity:
- sale_date x jan_code x customer_code
*/
WITH today_param AS (
    SELECT CAST(CONVERT(VARCHAR(8), GETDATE(), 112) AS INT) AS today_yyyymmdd
),
sales_base AS (
    SELECT
        RTRIM(d.sykd_scd) AS jan_code
      , d.sykd_uribi AS sale_date
      , RTRIM(d.sykd_tcd) AS customer_code
      , CAST(d.sykd_suryo AS DECIMAL(18,2)) AS sales_qty_raw
      , RTRIM(d.sykd_mei) AS detail_name
      , RTRIM(d.sykd_color) AS detail_color
      , RTRIM(d.sykd_size) AS detail_size
    FROM SYKD d
    CROSS JOIN today_param p
    WHERE RTRIM(d.sykd_scd) LIKE '452%'
      AND d.sykd_uribi <= p.today_yyyymmdd
      AND CAST(d.sykd_suryo AS DECIMAL(18,2)) <> 0
),
daily_customer_sales AS (
    SELECT
        f.sale_date
      , f.jan_code
      , f.customer_code
      , SUM(CASE WHEN f.sales_qty_raw > 0 THEN f.sales_qty_raw ELSE 0 END) AS sales_qty
      , SUM(CASE WHEN f.sales_qty_raw < 0 THEN ABS(f.sales_qty_raw) ELSE 0 END) AS return_qty
      , MAX(f.detail_name) AS detail_name
      , MAX(f.detail_color) AS detail_color
      , MAX(f.detail_size) AS detail_size
    FROM sales_base f
    GROUP BY
        f.sale_date
      , f.jan_code
      , f.customer_code
),
sms_attr AS (
    SELECT
        RTRIM(s.sms_scd) AS jan_code
      , MAX(RTRIM(s.sms_mei)) AS sms_name
      , MAX(RTRIM(s.sms_color)) AS sms_color
      , MAX(RTRIM(s.sms_size)) AS sms_size
    FROM SMS s
    WHERE RTRIM(s.sms_scd) LIKE '452%'
    GROUP BY RTRIM(s.sms_scd)
),
customer_attr AS (
    SELECT
        RTRIM(t.tms_tcd) AS customer_code
      , MAX(RTRIM(c.cms_mei1)) AS customer_name
    FROM TMS t
    LEFT JOIN CMS c
        ON t.tms_cmsid = c.cms_id
    GROUP BY RTRIM(t.tms_tcd)
)
SELECT
    d.jan_code AS jan_code
  , COALESCE(NULLIF(a.sms_name, ''), d.detail_name) AS item_name
  , COALESCE(NULLIF(a.sms_color, ''), d.detail_color) AS color
  , COALESCE(NULLIF(a.sms_size, ''), d.detail_size) AS size
  , RTRIM(CAST(d.sale_date AS VARCHAR(8))) AS sale_date
  , d.customer_code AS customer_code
  , ISNULL(c.customer_name, '') AS customer_name
  , d.sales_qty AS sales_qty
  , d.return_qty AS return_qty
FROM daily_customer_sales d
LEFT JOIN sms_attr a
    ON d.jan_code = a.jan_code
LEFT JOIN customer_attr c
    ON d.customer_code = c.customer_code
ORDER BY
    d.sale_date
  , d.customer_code
  , d.jan_code;
