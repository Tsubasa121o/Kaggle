/*
Purpose:
- Get active customers (code + name) for 452* sales analysis
Rule:
- Base period: trailing 365 days up to today
Granularity:
- customer_code
*/
WITH params AS (
    SELECT
        CAST(CONVERT(VARCHAR(8), GETDATE(), 112) AS INT) AS today_yyyymmdd
      , CAST(CONVERT(VARCHAR(8), DATEADD(DAY, -364, GETDATE()), 112) AS INT) AS from_yyyymmdd
),
sales_customers AS (
    SELECT
        RTRIM(d.sykd_tcd) AS customer_code
    FROM SYKD d
    CROSS JOIN params p
    WHERE RTRIM(d.sykd_scd) LIKE '452%'
      AND d.sykd_uribi BETWEEN p.from_yyyymmdd AND p.today_yyyymmdd
      AND CAST(d.sykd_suryo AS DECIMAL(18,2)) <> 0
    GROUP BY RTRIM(d.sykd_tcd)
),
customer_attr AS (
    SELECT
        RTRIM(t.tms_tcd) AS customer_code
      , MAX(RTRIM(c.cms_mei1)) AS customer_name
    FROM TMS t
    LEFT JOIN CMS c
        ON t.tms_cmsid = c.cms_id
    GROUP BY RTRIM(t.tms_tcd)
)
SELECT
    s.customer_code
  , ISNULL(c.customer_name, '') AS customer_name
FROM sales_customers s
LEFT JOIN customer_attr c
    ON s.customer_code = c.customer_code
ORDER BY
    s.customer_code;
