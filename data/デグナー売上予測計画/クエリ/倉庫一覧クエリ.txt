/*
Purpose:
- Warehouse list for 452* JAN analysis
- Definition-book compliant (no inferred column names)

Confirmed warehouse columns from handover definitions:
- ZMSN.zmsn_souko (current stock)
- ZMS.zms_souko (carry stock)
- SYKD.sykd_souko (sales/returns)
- NYKD.nykd_souko (receipts)

Output:
- warehouse_code
- warehouse_name_candidate (from EMSB.emsb_str)
- Per-table row counts / jan counts / qty summaries
*/
SET NOCOUNT ON;

DECLARE @from_yyyymmdd INT = 20081101;
DECLARE @to_yyyymmdd INT = CAST(CONVERT(VARCHAR(8), GETDATE(), 112) AS INT);

WITH zmsn_wh AS (
    SELECT
        NULLIF(RTRIM(z.zmsn_souko), '') AS warehouse_code,
        COUNT(*) AS zmsn_rows,
        COUNT(DISTINCT RTRIM(z.zmsn_scd)) AS zmsn_jan_count,
        SUM(CAST(z.zmsn_zai AS DECIMAL(18,2))) AS zmsn_current_qty
    FROM ZMSN z
    WHERE RTRIM(z.zmsn_scd) LIKE '452%'
    GROUP BY NULLIF(RTRIM(z.zmsn_souko), '')
),
zms_wh AS (
    SELECT
        NULLIF(RTRIM(z.zms_souko), '') AS warehouse_code,
        COUNT(*) AS zms_rows,
        COUNT(DISTINCT RTRIM(z.zms_scd)) AS zms_jan_count,
        SUM(CAST(z.zms_kuri AS DECIMAL(18,2))) AS zms_carry_qty
    FROM ZMS z
    WHERE RTRIM(z.zms_scd) LIKE '452%'
    GROUP BY NULLIF(RTRIM(z.zms_souko), '')
),
sykd_wh AS (
    SELECT
        NULLIF(RTRIM(d.sykd_souko), '') AS warehouse_code,
        COUNT(*) AS sykd_rows,
        COUNT(DISTINCT RTRIM(d.sykd_scd)) AS sykd_jan_count,
        SUM(CASE
                WHEN CAST(d.sykd_suryo AS DECIMAL(18,2)) > 0
                    THEN CAST(d.sykd_suryo AS DECIMAL(18,2))
                ELSE 0
            END) AS sykd_sales_qty,
        SUM(CASE
                WHEN CAST(d.sykd_suryo AS DECIMAL(18,2)) < 0
                    THEN ABS(CAST(d.sykd_suryo AS DECIMAL(18,2)))
                ELSE 0
            END) AS sykd_return_qty
    FROM SYKD d
    WHERE RTRIM(d.sykd_scd) LIKE '452%'
      AND d.sykd_uribi BETWEEN @from_yyyymmdd AND @to_yyyymmdd
      AND CAST(d.sykd_suryo AS DECIMAL(18,2)) <> 0
    GROUP BY NULLIF(RTRIM(d.sykd_souko), '')
),
nykd_wh AS (
    SELECT
        NULLIF(RTRIM(d.nykd_souko), '') AS warehouse_code,
        COUNT(*) AS nykd_rows,
        COUNT(DISTINCT RTRIM(d.nykd_scd)) AS nykd_jan_count,
        SUM(CASE
                WHEN CAST(d.nykd_suryo AS DECIMAL(18,2)) > 0
                    THEN CAST(d.nykd_suryo AS DECIMAL(18,2))
                ELSE 0
            END) AS nykd_receipt_qty
    FROM NYKD d
    JOIN NYKH h
        ON d.nykd_hid = h.nykh_id
    WHERE RTRIM(d.nykd_scd) LIKE '452%'
      AND h.nykh_uribi BETWEEN @from_yyyymmdd AND @to_yyyymmdd
    GROUP BY NULLIF(RTRIM(d.nykd_souko), '')
),
emsb_wh AS (
    SELECT
        NULLIF(RTRIM(e.emsb_souko), '') AS warehouse_code,
        MAX(NULLIF(RTRIM(e.emsb_str), '')) AS warehouse_name_candidate,
        MAX(NULLIF(RTRIM(e.emsb_kbn), '')) AS emsb_kbn
    FROM EMSB e
    GROUP BY NULLIF(RTRIM(e.emsb_souko), '')
),
all_wh AS (
    SELECT warehouse_code FROM zmsn_wh
    UNION
    SELECT warehouse_code FROM zms_wh
    UNION
    SELECT warehouse_code FROM sykd_wh
    UNION
    SELECT warehouse_code FROM nykd_wh
    UNION
    SELECT warehouse_code FROM emsb_wh
)
SELECT
    COALESCE(w.warehouse_code, '(BLANK)') AS warehouse_code,
    ISNULL(e.warehouse_name_candidate, '') AS warehouse_name_candidate,
    ISNULL(e.emsb_kbn, '') AS emsb_kbn,
    ISNULL(zn.zmsn_rows, 0) AS zmsn_rows,
    ISNULL(zn.zmsn_jan_count, 0) AS zmsn_jan_count,
    ISNULL(zn.zmsn_current_qty, 0) AS zmsn_current_qty,
    ISNULL(z.zms_rows, 0) AS zms_rows,
    ISNULL(z.zms_jan_count, 0) AS zms_jan_count,
    ISNULL(z.zms_carry_qty, 0) AS zms_carry_qty,
    ISNULL(s.sykd_rows, 0) AS sykd_rows,
    ISNULL(s.sykd_jan_count, 0) AS sykd_jan_count,
    ISNULL(s.sykd_sales_qty, 0) AS sykd_sales_qty,
    ISNULL(s.sykd_return_qty, 0) AS sykd_return_qty,
    ISNULL(n.nykd_rows, 0) AS nykd_rows,
    ISNULL(n.nykd_jan_count, 0) AS nykd_jan_count,
    ISNULL(n.nykd_receipt_qty, 0) AS nykd_receipt_qty
FROM all_wh w
LEFT JOIN emsb_wh e
    ON w.warehouse_code = e.warehouse_code
LEFT JOIN zmsn_wh zn
    ON w.warehouse_code = zn.warehouse_code
LEFT JOIN zms_wh z
    ON w.warehouse_code = z.warehouse_code
LEFT JOIN sykd_wh s
    ON w.warehouse_code = s.warehouse_code
LEFT JOIN nykd_wh n
    ON w.warehouse_code = n.warehouse_code
ORDER BY
    CASE WHEN w.warehouse_code IS NULL THEN 1 ELSE 0 END,
    w.warehouse_code;
