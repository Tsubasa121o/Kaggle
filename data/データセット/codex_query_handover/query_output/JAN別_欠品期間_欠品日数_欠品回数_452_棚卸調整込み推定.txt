/*
目的:
- 452始まりJANについて、棚卸調整を含めた欠品期間を推定する

主な出力:
- 欠品開始日
- 欠品終了日
- 欠品日数
- 欠品回数（JANごと）

前提:
- 在庫増減 = 入荷 + 返品 - 売上 + 棚卸調整
- 棚卸調整は得意先名（CMS）に「棚卸」を含む行で判定
- 棚卸調整の増減量は -sykd_suryo として扱う
- 振替は対象外
*/
WITH params AS (
    SELECT
        20130101 AS from_yyyymmdd,
        CAST(CONVERT(VARCHAR(8), GETDATE(), 112) AS INT) AS to_yyyymmdd,
        CAST(GETDATE() AS DATE) AS today_date
),
item_attr AS (
    SELECT
        RTRIM(s.sms_scd) AS jan_code,
        MAX(RTRIM(s.sms_mei)) AS item_name
    FROM SMS s
    WHERE RTRIM(s.sms_scd) LIKE '452%'
    GROUP BY RTRIM(s.sms_scd)
),
sales_classified AS (
    SELECT
        RTRIM(d.sykd_scd) AS jan_code,
        d.sykd_uribi AS yyyymmdd,
        CAST(d.sykd_suryo AS DECIMAL(18,2)) AS qty_raw,
        ISNULL(RTRIM(c.cms_mei1), '') AS partner_name
    FROM SYKD d
    LEFT JOIN TMS t
      ON RTRIM(d.sykd_tcd) = RTRIM(t.tms_tcd)
    LEFT JOIN CMS c
      ON t.tms_cmsid = c.cms_id
    CROSS JOIN params p
    WHERE RTRIM(d.sykd_scd) LIKE '452%'
      AND d.sykd_uribi BETWEEN p.from_yyyymmdd AND p.to_yyyymmdd
      AND CAST(d.sykd_suryo AS DECIMAL(18,2)) <> 0
),
sales_daily AS (
    SELECT
        s.jan_code,
        s.yyyymmdd,
        SUM(
            CASE
                WHEN s.partner_name LIKE N'%棚卸%' THEN 0
                WHEN s.qty_raw > 0 THEN s.qty_raw
                ELSE 0
            END
        ) AS sales_qty,
        SUM(
            CASE
                WHEN s.partner_name LIKE N'%棚卸%' THEN 0
                WHEN s.qty_raw < 0 THEN ABS(s.qty_raw)
                ELSE 0
            END
        ) AS return_qty,
        SUM(
            CASE
                WHEN s.partner_name LIKE N'%棚卸%'
                    THEN -s.qty_raw
                ELSE 0
            END
        ) AS adjust_qty
    FROM sales_classified s
    GROUP BY
        s.jan_code,
        s.yyyymmdd
),
receipt_daily AS (
    SELECT
        RTRIM(d.nykd_scd) AS jan_code,
        h.nykh_uribi AS yyyymmdd,
        SUM(CASE WHEN CAST(d.nykd_suryo AS DECIMAL(18,2)) > 0
                 THEN CAST(d.nykd_suryo AS DECIMAL(18,2))
                 ELSE 0 END) AS receipt_qty
    FROM NYKD d
    JOIN NYKH h
      ON d.nykd_hid = h.nykh_id
    CROSS JOIN params p
    WHERE RTRIM(d.nykd_scd) LIKE '452%'
      AND h.nykh_uribi BETWEEN p.from_yyyymmdd AND p.to_yyyymmdd
    GROUP BY
        RTRIM(d.nykd_scd),
        h.nykh_uribi
),
daily_movement AS (
    SELECT
        COALESCE(s.jan_code, r.jan_code) AS jan_code,
        COALESCE(s.yyyymmdd, r.yyyymmdd) AS yyyymmdd,
        ISNULL(s.sales_qty, 0) AS sales_qty,
        ISNULL(s.return_qty, 0) AS return_qty,
        ISNULL(s.adjust_qty, 0) AS adjust_qty,
        ISNULL(r.receipt_qty, 0) AS receipt_qty,
        ISNULL(r.receipt_qty, 0)
            - ISNULL(s.sales_qty, 0)
            + ISNULL(s.return_qty, 0)
            + ISNULL(s.adjust_qty, 0) AS net_qty
    FROM sales_daily s
    FULL OUTER JOIN receipt_daily r
      ON s.jan_code = r.jan_code
     AND s.yyyymmdd = r.yyyymmdd
),
current_stock AS (
    SELECT
        RTRIM(z.zmsn_scd) AS jan_code,
        SUM(CAST(z.zmsn_zai AS DECIMAL(18,2))) AS current_qty
    FROM ZMSN z
    WHERE RTRIM(z.zmsn_scd) LIKE '452%'
    GROUP BY RTRIM(z.zmsn_scd)
),
event_stock AS (
    SELECT
        m.jan_code,
        m.yyyymmdd,
        m.receipt_qty,
        m.adjust_qty,
        SUM(m.net_qty) OVER (
            PARTITION BY m.jan_code
            ORDER BY m.yyyymmdd
            ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
        ) AS cum_net_qty,
        SUM(m.net_qty) OVER (PARTITION BY m.jan_code) AS total_net_qty,
        ISNULL(c.current_qty, 0) AS current_qty
    FROM daily_movement m
    LEFT JOIN current_stock c
      ON m.jan_code = c.jan_code
),
event_stock_with_level AS (
    SELECT
        e.jan_code,
        e.yyyymmdd,
        e.receipt_qty,
        e.adjust_qty,
        e.current_qty - e.total_net_qty + e.cum_net_qty AS stock_est
    FROM event_stock e
),
event_stock_with_lag AS (
    SELECT
        e.jan_code,
        e.yyyymmdd,
        e.receipt_qty,
        e.adjust_qty,
        e.stock_est,
        LAG(e.stock_est) OVER (
            PARTITION BY e.jan_code
            ORDER BY e.yyyymmdd
        ) AS prev_stock_est
    FROM event_stock_with_level e
),
stockout_starts AS (
    SELECT
        s.jan_code,
        s.yyyymmdd AS start_yyyymmdd
    FROM event_stock_with_lag s
    WHERE s.stock_est <= 0
      AND COALESCE(s.prev_stock_est, 1) > 0
),
receipt_events AS (
    SELECT
        jan_code,
        yyyymmdd AS receipt_yyyymmdd
    FROM receipt_daily
    WHERE receipt_qty > 0
),
stockout_periods AS (
    SELECT
        st.jan_code,
        st.start_yyyymmdd,
        nxt.receipt_yyyymmdd AS end_yyyymmdd
    FROM stockout_starts st
    OUTER APPLY (
        SELECT TOP 1 r.receipt_yyyymmdd
        FROM receipt_events r
        WHERE r.jan_code = st.jan_code
          AND r.receipt_yyyymmdd > st.start_yyyymmdd
        ORDER BY r.receipt_yyyymmdd
    ) nxt
),
adjust_summary AS (
    SELECT
        d.jan_code,
        SUM(d.adjust_qty) AS total_adjust_qty
    FROM daily_movement d
    GROUP BY d.jan_code
)
SELECT
    p.jan_code AS JANコード,
    ISNULL(i.item_name, '') AS 品名,
    RTRIM(CAST(p.start_yyyymmdd AS VARCHAR(8))) AS 欠品開始日,
    CASE
        WHEN p.end_yyyymmdd IS NULL THEN ''
        ELSE RTRIM(CAST(p.end_yyyymmdd AS VARCHAR(8)))
    END AS 欠品終了日,
    DATEDIFF(
        DAY,
        TRY_CONVERT(DATE, CONVERT(VARCHAR(8), p.start_yyyymmdd), 112),
        COALESCE(
            TRY_CONVERT(DATE, CONVERT(VARCHAR(8), p.end_yyyymmdd), 112),
            prm.today_date
        )
    ) AS 欠品日数,
    ROW_NUMBER() OVER (
        PARTITION BY p.jan_code
        ORDER BY p.start_yyyymmdd
    ) AS 欠品連番,
    COUNT(*) OVER (
        PARTITION BY p.jan_code
    ) AS 欠品回数,
    ISNULL(a.total_adjust_qty, 0) AS 棚卸調整数量合計,
    CASE
        WHEN p.end_yyyymmdd IS NULL THEN N'継続中'
        ELSE N'解消'
    END AS 欠品状態
FROM stockout_periods p
LEFT JOIN item_attr i
  ON p.jan_code = i.jan_code
LEFT JOIN adjust_summary a
  ON p.jan_code = a.jan_code
CROSS JOIN params prm
ORDER BY
    p.jan_code,
    p.start_yyyymmdd;
